多渠道流量（Web/企微/TG/WA/小红书等）通过 Webhook/WebSocket 接入后，打入消息中台（MessageHub），最终转化为统一收件箱（InboxService）。【Go 源码落地：多渠道收件箱适配器】gopackage main

import (
	"context"
	"encoding/json"
	"time"
)

// MessageHub 统一收件箱的唯一消息标准
type MessageEvent struct {
	EventID   string    `json:"event_id"`   // 全局唯一事件ID
	SessionID string    `json:"session_id"` // 系统内部映射的唯一会话ID
	Channel   string    `json:"channel"`    // 渠道来源: "web", "telegram", "whatsapp", "xhs"
	SenderID  string    `json:"sender_id"`  // 最终客户的唯一物理标识（如手机号、TG UserID）
	MsgType   string    `json:"msg_type"`   // "text", "image", "file"
	Content   string    `json:"content"`    // 纯文本消息内容
	Timestamp time.Time `json:"timestamp"`
}

// InboxService 接收端，负责高并发防抖和排队
type InboxService struct {
	redisClient *RedisClient
}

func (i *InboxService) HandleIngressMessage(ctx context.Context, event MessageEvent) error {
	// 关键卡位 1：检测会话是否已被商户后台锁定为“人工接管”状态
	humanLockKey := "hivemtk:lock:human:" + event.SessionID
	if isHuman, _ := i.redisClient.Get(humanLockKey); isHuman == "true" {
		return SaveToPostgresOnly(event) // 仅存入数据库供人类查看，阻断 AI 路由
	}

	// 关键卡位 2：利用 Redis 锁进行高并发串行化排队防抖（锁 15 秒）
	aiLockKey := "hivemtk:lock:ai_processing:" + event.SessionID
	success, _ := i.redisClient.SetNX(aiLockKey, "busy", 15*time.Second)
	
	if !success {
		// 如果前一秒用户刚发过消息且 AI 正在推理，将新消息推入 Redis 临时队列进行“上下文追加”
		pendingKey := "hivemtk:pending:" + event.SessionID
		_ = i.redisClient.LPush(pendingKey, event.Content)
		return nil
	}

	// 成功获取会话锁，触发智能体运行时（AgentRuntime）
	go i.triggerAgentRuntime(event)
	return nil
}
