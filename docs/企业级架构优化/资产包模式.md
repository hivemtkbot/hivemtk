二、 符合 OpenAI 协议的 Go 后端动态“织布”算法实现在商户本地的 Go 后端服务中，既然资产包就是标准的 messages 数组，那么动态拼装的算法就会变得极其干净和标准。商户自己本地独享的 RAG 知识库检索结果、以及从 Redis 中读取的 客户真实聊天历史记录，必须作为符合协议规范的系统切片和上下文切片，动态插入（Splice）到资产包消息数组的特定物理位置中：

package main

import (
	"fmt"
)

// 符合 OpenAI / Ollama 标准协议的消息结构体
type Message struct {
	Role    string `json:"role"`    // "system", "user", "assistant"
	Content string `json:"content"` // 纯文本内容
}

// 最终本地 PostgreSQL 数据库 ORM 模型定义
type HivePureAsset struct {
	AssetID   string    `gorm:"type:varchar(64);uniqueIndex"`
	Title     string    `gorm:"type:varchar(128)"`
	Messages  []Message `gorm:"type:jsonb"` // 100% 遵守 OpenAI 协议的 messages 数组
}

// 标准 OpenAI 协议下的全量 Prompt 动态织布算法
func WeaveOpenAICompatibleChain(
	userQuery string, 
	asset HivePureAsset, 
	localMerchantDocs []string, // 商户本地检索出来的 RAG 文档切片
	redisChatHistory []Message, // 从 Redis 中读取的最近 3 轮活跃聊天明文历史
) []Message {
	// 建立一个全新的、用于最终请求大模型容器的切片
	var finalPayload []Message

	// 【第一步：全量继承资产包本身的控制链（第 0 项是全局 System 销冠人设，其余项是原生话术 Few-Shots）】
	if len(asset.Messages) > 0 {
		finalPayload = append(finalPayload, asset.Messages...)
	}

	// 【第二步：动态插入商户本地自建的 RAG 知识库内容】
	// 遵照 OpenAI 协议：知识库作为一条补充的本地事实背景，以 system 角色动态追加在资产包的原始人设之后
	if len(localMerchantDocs) > 0 {
		var ragContext string
		for i, doc := range localMerchantDocs {
			ragContext += fmt.Sprintf("[商户本地官方参数 %d]: %s\n", i+1, doc)
		}
		
		ragSystemMessage := Message{
			Role:    "system",
			Content: fmt.Sprintf("# 实时验证知识库上下文\n请结合以下商户本地的实时参考数据，精准、诚实地回答用户后续的提问，不要凭空编造事实：\n\n%s", ragContext),
		}
		
		// 优雅插入到数组中：紧跟在资产包原生 System 消息的最前方或 Few-Shots 的交界处
		// 这里我们直接追加在资产包基本指令链之后，使其作为 RAG 引导语
		finalPayload = append(finalPayload, ragSystemMessage)
	}

	// 【第三步：追加最终 Session 的活跃历史聊天记录】
	// 保持协议上下文的连贯性
	if len(redisChatHistory) > 0 {
		finalPayload = append(finalPayload, redisChatHistory...)
	}

	// 【第四步：追加当前用户最新的真实提问（触发临门一脚推理）】
	finalMessages := append(finalPayload, Message{
		Role:    "user",
		Content: userQuery,
	})

	return finalMessages
}
