// 1. 触发转人工熔断锁（由后台不可见 Agent 状态机动态拦截触发）
func (h *HumanEscalationManager) TriggerCensorshipEscalation(ctx context.Context, sessionID string, reason string) error {
	// 下发 Redis 分布式协同锁，锁定当前会话为人工状态，防止 AI 继续调用算力
	lockKey := fmt.Sprintf("hivemtk:lock:human:%s", sessionID)
	_, err := h.redisClient.Set(ctx, lockKey, "true", 0) // 0 代表永久锁定，直到人工点击解决
	if err != nil {
		return err
	}

	// 记录触发转人工的核心原因（如：危机感打分过高、客户情绪极度愤怒）
	reasonKey := fmt.Sprintf("hivemtk:session:escalate_reason:%s", sessionID)
	_, _ = h.redisClient.Set(ctx, reasonKey, reason, 24*time.Hour)

	// 将转人工事件推送至商户异步通知总线，触发 Vue 3 工作台长连接发出“叮咚”弹窗警报
	notificationPayload := map[string]string{
		"event":      "TRANSFER_TO_HUMAN",
		"session_id": sessionID,
		"reason":     reason,
		"timestamp":  time.Now().Format(time.RFC3339),
	}
	jsonBytes, _ := json.Marshal(notificationPayload)
	_ = h.redisClient.LPush(ctx, "hivemtk:queue:merchant_notif", string(jsonBytes))

	fmt.Printf("🚨 [门禁熔断] 会话 [%s] 触发高危危机感，成功下发分布式锁，AI 已安全闭嘴。\n", sessionID)
	return nil
}

// 2. 检查会话状态机（由前台可见 Web 接入层核心路由高频调用，用作大模型调用前的“前置阻断器”）
func (h *HumanEscalationManager) IsSessionLockedForHuman(ctx context.Context, sessionID string) bool {
	lockKey := fmt.Sprintf("hivemtk:lock:human:%s", sessionID)
	val, err := h.redisClient.Get(ctx, lockKey)
	if err != nil {
		return false // 缓存降级安全策略
	}
	return val == "true"
}
