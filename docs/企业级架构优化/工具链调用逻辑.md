大模型（LLM）本身不具备查询数据库、修改订单或发送外部消息的能力。hivemtk 采用 “思维-行动（Thought-Action）”的反向拦截拦截范式，通过 Go 后端赋予大模型“手脚”。1. 工具调用的双向拦截设计在工具调用的模式下，Go 后端后台守护进程（agent_daemon）需要执行 “两次组装、两次大模型推理” 的闭环逻辑。其核心拦截规则如下：

                    【 第一次大模型推理 (Stream) 】
                                  │
                                  ▼
             [ 吐字流式 Chunk 通过 Go 内存滑动窗口缓冲区 ]
                                  │
      ┌───────────────────────────┴───────────────────────────┐
      ▼ (检测到字符包含 "调用工具：")                            ▼ (未检测到工具块)
[ 状态机立即激活熔断 (掐断向前端转发) ]                 [ 正常拟人化文本回复 ]
      │                                                       │
      ▼ (Go 在内存中默默吞下后面的 JSON)                        ▼
[ 解析出 JSON 参数 -> 调度工具注册表 ]                  [ 经过正则裁剪器切除尾巴 ]
      │                                                       │
      ▼ (执行本地 Go 函数查询/修改业务)                         ▼
[ 拿到工具真实结果 -> 执行‘二进宫’二次组装 ]              [ 纯净文本秒级发送给最终用户 ]


2. 工具注册表调用中枢代码实现 (Tool Registry Router)在 Go 后端中，工具注册表通过一个高效的路由中枢统一管理，该中枢集成了权限审计、频次限制、异常重试、以及成本监控四大核心生产级防线：gopackage main

import (
	"encoding/json"
	"fmt"
)

// 工具调用标准返回结构
type ToolResponse struct {
	Status string `json:"status"` // "success" 或 "error"
	Data   string `json:"data"`   // 给大模型看的真实结构化文本结果
}

// 工具注册表全局路由中心（对应架构图中的 30+ 工具调度手脚）
func RouteToToolRegistry(toolName string, parameters map[string]interface{}) ToolResponse {
	// 【生产级防线 1：统一权限与合规审计】
	fmt.Printf("🛡️ [工具审计] 正在调用工具: %s, 传入参数: %v\n", toolName, parameters)

	// 【生产级防线 2：安全限流与频次拦截】
	if isRateLimited(toolName) {
		return ToolResponse{Status: "error", Data: "系统繁忙，该工具调用过于频繁，请稍后再试。"}
	}

	// 核心业务路由分发
	switch toolName {
	case "customer.get_profile":
		// 调用客户模块：从本地 PostgreSQL 获取该潜客的特征
		return ToolResponse{Status: "success", Data: "{\"user_name\": \"张三\", \"level\": \"VIP\", \"pref\": \"蓝莓味\"}"}

	case "rag.search_knowledge":
		// 调用知识库检索：获取商户本地私有知识库的产品说明
		query, _ := parameters["query"].(string)
		return ExecutePgVectorSearch(query)

	case "reach.send_whatsapp":
		// 调用营销触达：利用 Meta 官方 API 主动发送营销优惠券卡片
		phone, _ := parameters["phone"].(string)
		couponID, _ := parameters["coupon_id"].(string)
		return TriggerWhatsAppTemplateAPI(phone, couponID)

	default:
		// 【生产级防线 3：异常容错拦截】
		return ToolResponse{Status: "error", Data: fmt.Sprintf("未找到名为 [%s] 的工具组件，请检查工具注册表。", toolName)}
	}
}

// 模拟桩函数
func isRateLimited(t string) bool { return false }
func ExecutePgVectorSearch(q string) ToolResponse { return ToolResponse{Status: "success", Data: "蓝莓味库存500"} }
func TriggerWhatsAppTemplateAPI(p, c string) ToolResponse { return ToolResponse{Status: "success", Data: "已发送"} }
